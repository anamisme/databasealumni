<script>
// ══════════════════════════════════════════════════════════════
//  ⚙️  KONFIGURASI LEMBAGA
//  - Ganti gid dengan GID sheet masing-masing lembaga
//  - GID bisa dilihat di URL spreadsheet: ...#gid=ANGKA
//  - warna: kode hex untuk identitas warna tiap lembaga
// ══════════════════════════════════════════════════════════════
const SPREADSHEET_ID = "2PACX-1vTghKGApa4yXg6ckCH30JZzG0T1YICx2qE74v386oEFIHrQLfVMxSBboMAQYpvdee7cZeW-tGTuKxgQ";

const LEMBAGA = [
  {
    nama:    "MTs Baitul Hikmah",
    singkat: "MTS",
    gid:     "0",
    warna:   "#4f46e5",     // indigo
  },
  {
    nama:    "MIS Baitul Hikmah",
    singkat: "MIS",
    gid:     "451330348",
    warna:   "#0891b2",          // cyan
  },
  {
    nama:    "PKBM Baitul Hikmah",
    singkat: "PKBM",
    gid:     "2060155617",
    warna:   "#059669",          // emerald
  },
  {
    nama:    "KB Baitul Hikmah",
    singkat: "KB",
    gid:     "907882709",
    warna:   "#d97706",          // amber
  },
];

// ── Mapping nama kolom (sesuaikan jika berbeda di sheet) ──────
const KOLOM = {
  nama:     "Nama",
  nisn:     "NISN",
  status:   "Status",       // nilai: aktif / alumni
  jenjang:  "Jenjang",
  kelas:    "Kelas",
  masuk:    "Tahun_Masuk",
  lulus:    "Tahun_Lulus",
  prestasi: "Prestasi",     // pisah beberapa dengan |
  gender:   "Jenis_Kelamin", // L / P
};
// ══════════════════════════════════════════════════════════════

let allData       = [];
let currentFilter = "semua";
let currentQuery  = "";
let currentIdx    = 0;
let cache         = {};

function csvUrl(gid) {
  return `https://docs.google.com/spreadsheets/d/e/${SPREADSHEET_ID}/pub?gid=${gid}&single=true&output=csv`;
}

// ── Build lembaga tabs ────────────────────────────────────────
function buildTabs() {
  const wrap = document.getElementById("lembagaTabs");
  wrap.innerHTML = "";
  LEMBAGA.forEach((l, i) => {
    const btn = document.createElement("button");
    btn.className = "lembaga-tab" + (i === 0 ? " active" : "");
    btn.innerHTML = `<span class="dot" style="background:${l.warna}"></span>${l.singkat}`;
    btn.addEventListener("click", () => switchLembaga(i));
    wrap.appendChild(btn);
  });
}

function switchLembaga(idx) {
  currentIdx = idx;
  currentQuery = "";
  document.getElementById("searchInput").value = "";

  document.querySelectorAll(".lembaga-tab").forEach((t, i) => {
    t.classList.toggle("active", i === idx);
  });

  const l = LEMBAGA[idx];
  document.getElementById("lembagaDot").style.background = l.warna;
  document.getElementById("lembagaName").textContent = l.nama;
  document.getElementById("lembagaSub").textContent = "Memuat…";
  document.getElementById("statTotal").textContent = "–";
  document.getElementById("statAktif").textContent = "–";
  document.getElementById("statAlumni").textContent = "–";
  document.getElementById("listArea").innerHTML =
    `<div class="state-box"><div class="loader"></div><p style="margin-top:1rem">Memuat data ${escHtml(l.nama)}…</p></div>`;

  if (cache[idx]) {
    allData = cache[idx];
    updateStats();
    render();
    return;
  }

  Papa.parse(csvUrl(l.gid), {
    download: true, header: true, skipEmptyLines: true,
    complete(res) {
      allData = res.data.map(r => ({
        nama:     (r[KOLOM.nama]     || "").trim(),
        nisn:     (r[KOLOM.nisn]     || "").trim(),
        status:   (r[KOLOM.status]   || "").trim().toLowerCase(),
        jenjang:  (r[KOLOM.jenjang]  || "").trim(),
        kelas:    (r[KOLOM.kelas]    || "").trim(),
        masuk:    (r[KOLOM.masuk]    || "").trim(),
        lulus:    (r[KOLOM.lulus]    || "-").trim(),
        prestasi: (r[KOLOM.prestasi] || "").trim().split("|").map(s=>s.trim()).filter(Boolean),
        gender:   parseGender(r[KOLOM.gender] || ""),
      })).filter(r => r.nama);
      cache[idx] = allData;
      updateStats();
      render();
    },
    error() {
      document.getElementById("lembagaSub").textContent = "Gagal memuat";
      document.getElementById("listArea").innerHTML = `
        <div class="state-box">
          <div class="icon">⚠️</div>
          <h3>Gagal memuat data</h3>
          <p>Pastikan GID sheet <strong>${escHtml(l.singkat)}</strong> sudah benar dan sheet sudah dipublikasikan.</p>
        </div>`;
    }
  });
}

function parseGender(val) {
  const v = val.trim().toLowerCase();
  if (v === "p" || v.includes("perempuan") || v.includes("wanita")) return "P";
  return "L";
}

function updateStats() {
  const l = LEMBAGA[currentIdx];
  document.getElementById("statTotal").textContent  = allData.length;
  document.getElementById("statAktif").textContent  = allData.filter(s => s.status === "aktif").length;
  document.getElementById("statAlumni").textContent = allData.filter(s => s.status === "alumni").length;
  document.getElementById("lembagaSub").textContent = allData.length + " siswa terdaftar";
}

function render() {
  const q     = currentQuery.toLowerCase().trim();
  let hasil   = allData;
  if (currentFilter === "aktif")  hasil = hasil.filter(s => s.status === "aktif");
  if (currentFilter === "alumni") hasil = hasil.filter(s => s.status === "alumni");
  if (q) hasil = hasil.filter(s => s.nama.toLowerCase().includes(q));

  const meta = document.getElementById("resultsMeta");
  const list = document.getElementById("listArea");

  if (!q) {
    meta.innerHTML = "";
    list.innerHTML = `
      <div class="state-box">
        <div class="icon">🔍</div>
        <h3>Cari nama siswa</h3>
        <p>Masukkan nama untuk menemukan data siswa ${escHtml(LEMBAGA[currentIdx].nama)}.</p>
      </div>`;
    return;
  }

  meta.innerHTML = `Ditemukan <strong>${hasil.length} siswa</strong> untuk "<strong>${escHtml(currentQuery)}</strong>"`;

  if (!hasil.length) {
    list.innerHTML = `
      <div class="state-box">
        <div class="icon">😕</div>
        <h3>Tidak ditemukan</h3>
        <p>Coba gunakan nama lengkap atau ejaan yang berbeda.</p>
      </div>`;
    return;
  }

  list.innerHTML = hasil.map(s => cardHTML(s)).join("");
  list.querySelectorAll(".siswa-card").forEach(c => {
    c.addEventListener("click", () => c.classList.toggle("open"));
  });
}

function initials(nama) {
  return nama.split(" ").slice(0,2).map(w=>w[0]||"").join("").toUpperCase();
}
function escHtml(s) {
  return String(s).replace(/&/g,"&amp;").replace(/</g,"&lt;").replace(/>/g,"&gt;");
}

function cardHTML(s) {
  const lu = s.status === "aktif" ? "Belum lulus" : (s.lulus || "–");
  const pr = s.prestasi.length
    ? `<div class="prestasi-section"><label>Prestasi</label>${s.prestasi.map(p=>`<div class="prestasi-item"><span>🏅</span>${escHtml(p)}</div>`).join("")}</div>`
    : "";
  return `
    <div class="siswa-card" role="button" tabindex="0">
      <div class="card-main">
        <div class="avatar av-${s.gender==="P"?"p":"l"}">${initials(s.nama)}</div>
        <div class="card-info">
          <div class="card-nama">${escHtml(s.nama)}</div>
          <div class="card-badges">
            <span class="badge badge-${s.status==="aktif"?"aktif":"alumni"}">${s.status==="aktif"?"Aktif":"Alumni"}</span>
            ${s.jenjang?`<span class="badge badge-gray">${escHtml(s.jenjang)}</span>`:""}
            ${s.kelas&&s.kelas!=="-"?`<span class="badge badge-gray">${escHtml(s.kelas)}</span>`:""}
          </div>
        </div>
        <svg class="card-chevron" width="20" height="20" fill="none" viewBox="0 0 24 24" stroke="currentColor">
          <polyline points="6 9 12 15 18 9" stroke-width="2" stroke-linecap="round" stroke-linejoin="round"/>
        </svg>
      </div>
      <div class="card-detail">
        <div class="detail-grid">
          <div class="detail-item"><label>Nama Lengkap</label><span>${escHtml(s.nama)}</span></div>
          <div class="detail-item"><label>NISN</label><span>${escHtml(s.nisn)||"–"}</span></div>
          <div class="detail-item"><label>Jenjang</label><span>${escHtml(s.jenjang)||"–"}</span></div>
          <div class="detail-item"><label>Kelas / Rombel</label><span>${escHtml(s.kelas)||"–"}</span></div>
          <div class="detail-item"><label>Tahun Masuk</label><span>${escHtml(s.masuk)||"–"}</span></div>
          <div class="detail-item"><label>Tahun Lulus</label><span>${escHtml(lu)}</span></div>
        </div>
        ${pr}
      </div>
    </div>`;
}

// ── Events ────────────────────────────────────────────────────
document.getElementById("searchInput").addEventListener("input", e => {
  currentQuery = e.target.value; render();
});
document.getElementById("clearBtn").addEventListener("click", () => {
  const inp = document.getElementById("searchInput");
  inp.value = ""; currentQuery = ""; render(); inp.focus();
});
document.querySelectorAll(".pill").forEach(btn => {
  btn.addEventListener("click", () => {
    document.querySelectorAll(".pill").forEach(p=>p.classList.remove("active"));
    btn.classList.add("active");
    currentFilter = btn.dataset.filter;
    render();
  });
});
document.getElementById("listArea").addEventListener("keydown", e => {
  if (e.key==="Enter"||e.key===" ") {
    const c = e.target.closest(".siswa-card");
    if (c) { e.preventDefault(); c.classList.toggle("open"); }
  }
});

// ── Init ──────────────────────────────────────────────────────
buildTabs();
switchLembaga(0);
</script>
</body>
